import React, { useState, useEffect } from 'react';
import { View, Text, Button, ScrollView, StyleSheet } from 'react-native';
import Voice from '@react-native-voice/voice';

export default function VoiceLogEntry() {
  const [isListening, setIsListening] = useState(false);
  const [transcript, setTranscript] = useState('');
  const [logEntries, setLogEntries] = useState([]);
  const [fieldIndex, setFieldIndex] = useState(0);

  const fields = [
    'Patient age and sex',
    'ASA grade',
    'BMI',
    'Procedure name and specialty',
    'Anaesthesia type',
    'Airway device used',
    'Drugs administered',
    'Monitoring used',
    'Complications',
    'Supervision level',
    'Reflection or learning point',
  ];

  useEffect(() => {
    Voice.onSpeechResults = onSpeechResults;
    Voice.onSpeechError = onSpeechError;
    return () => {
      Voice.destroy().then(Voice.removeAllListeners);
    };
  }, []);

  const onSpeechResults = (event) => {
    if (event.value && event.value.length > 0) {
      setTranscript(event.value[0]);
    }
  };

  const onSpeechError = (event) => {
    console.error('Speech recognition error:', event.error);
    setTranscript('');
    setIsListening(false);
  };

  const startListening = async () => {
    try {
      await Voice.start('en-US');
      setIsListening(true);
      setTranscript('');
    } catch (e) {
      console.error(e);
    }
  };

  const stopListening = async () => {
    try {
      await Voice.stop();
      setIsListening(false);
      // Save the transcript as the current field's input
      const newLogEntries = [...logEntries];
      newLogEntries[fieldIndex] = transcript;
      setLogEntries(newLogEntries);
      setTranscript('');
      if (fieldIndex < fields.length - 1) {
        setFieldIndex(fieldIndex + 1);
      }
    } catch (e) {
      console.error(e);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.header}>Anaesthetic Logbook Voice Entry</Text>
      <Text style={styles.fieldLabel}>Current Field:</Text>
      <Text style={styles.fieldName}>{fields[fieldIndex]}</Text>

      <ScrollView style={styles.transcriptBox}>
        <Text style={styles.transcriptText}>{transcript || 'Speak now...'}</Text>
      </ScrollView>

      <View style={styles.buttonRow}>
        {!isListening ? (
          <Button title="Start Recording" onPress={startListening} />
        ) : (
          <Button title="Stop Recording" onPress={stopListening} color="red" />
        )}
      </View>

      <View style={styles.logPreview}>
        <Text style={styles.previewHeader}>Entry Preview:</Text>
        {fields.map((field, idx) => (
          <Text key={idx} style={styles.previewText}>
            {field}: {logEntries[idx] || '...'}
          </Text>
        ))}
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { padding: 20, flex: 1 },
  header: { fontSize: 22, fontWeight: 'bold', marginBottom: 20 },
  fieldLabel: { fontSize: 16 },
  fieldName: { fontSize: 20, marginBottom: 10, color: 'blue' },
  transcriptBox: {
    borderColor: '#ccc',
    borderWidth: 1,
    padding: 10,
    height: 100,
    marginBottom: 20,
  },
  transcriptText: { fontSize: 18 },
  buttonRow: { marginBottom: 20 },
  logPreview: { flex: 1 },
  previewHeader: { fontSize: 18, fontWeight: 'bold', marginBottom: 10 },
  previewText: { fontSize: 16, marginBottom: 5 },
});
